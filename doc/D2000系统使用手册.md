# CAN

此项目开启两个CAN控制器，CAN 0-1，在LINUX操作系统采用的socket CAN的方式进行通信。
使用操作系统指令设置CAN波特率和采样参数。

## 检查系统CAN

LINUX指令如下：
```shell
ifconfig -a
```
如果出现下图，说明系统中有can驱动加载了，并登记can设备。

## 依赖库开发应用程序
交叉编译工具链中没有支持lib
- [ ] can-utils 
- [ ] libsocketcan

使用libsocketcan 开发can应用

```c
#include <stdio.h>

#include <stdlib.h>

#include <string.h>

#include <unistd.h>

#include <net/if.h>

#include <sys/ioctl.h>

#include <sys/socket.h>

#include <libsocketcan.h>

#include <linux/can.h>

#include <linux/can/raw.h>

  

#define CAN_INTERFACE "can1"

#define BITRATE 500000

  

int main() {

    int ret;

    int sock;

    struct sockaddr_can addr;

    struct ifreq ifr;

    struct can_frame frame;

    int count;

  

    // 1. 检查并设置 CAN 接口

    printf("Setting up CAN interface...\n");

    // 检查接口是否存在

    ret = can_do_stop(CAN_INTERFACE);

    if (ret < 0) {

        printf("CAN interface %s not found or error: %s\n",

               CAN_INTERFACE, strerror(ret));

        return -1;

    }

    // 设置波特率

    ret = can_set_bitrate(CAN_INTERFACE, BITRATE);

    if (ret < 0) {

        printf("Failed to set bitrate: %s\n", strerror(ret));

        return -1;

    }

    // 启用 CAN 接口

    ret = can_do_start(CAN_INTERFACE);

    if (ret < 0) {

        printf("Failed to start CAN interface: %s\n", strerror(ret));

        return -1;

    }

    printf("CAN interface %s started at %d bps\n", CAN_INTERFACE, BITRATE);

    // 创建套接字

    sock = socket(PF_CAN, SOCK_RAW, CAN_RAW);

    strcpy(ifr.ifr_name, "can1");

    ioctl(sock, SIOCGIFINDEX, &ifr);

    addr.can_family = AF_CAN;

    addr.can_ifindex = ifr.ifr_ifindex;

    bind(sock, (struct sockaddr *)&addr, sizeof(addr));

    // 设置接收超时

    struct timeval tv = {1, 0}; // 1秒

    setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));

    // 发送 CAN 帧

    frame.can_id = 0x123;

    frame.can_dlc = 8;

    memset(frame.data, 0xAA, 8);

    write(sock, &frame, sizeof(frame));

    // 接收 CAN 帧

    while(count < 10) {

        int nbytes = read(sock, &frame, sizeof(frame));

        if(nbytes > 0) {

            printf("收到帧 ID: 0x%X 数据: ", frame.can_id);

            for(int i = 0; i < frame.can_dlc; i++)

                printf("%02X ", frame.data[i]);

            printf("\n");

        }

        count++;

    }

    close(sock);

  

    can_do_stop(CAN_INTERFACE);

    printf("CAN interface stopped\n");

    return 0;

}

```

ip link set up can1
