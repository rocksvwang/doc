# U-BOOT 启动参数

## 文件系统在硬盘

```shell
setenv bootcmd 'run  boot_kylin_cmd'
setenv boot_kylin_cmd 'run load_kernel;run load_dtb;bootm 0x90200000 - 0x90100000'                   
setenv load_kernel 'hqflashrd 0x90200000 0x600000 0x1406000'
setenv load_dtb 'hqflashrd 0x90100000 0x500000 0x3000'
setenv bootargs 'console=ttyAMA1,115200 earlycon=pl011,0x28001000 root=/dev/nvme0n1p3 rootwait rw'   
printenv
saveenv
```

## 文件系统在norflash

```shell
setenv bootcmd 'run  boot_kylin_cmd'
setenv boot_kylin_cmd 'run load_kernel;run load_dtb;run load_fs;bootm 0x90200000 0x93000000 0x90100000 '
setenv load_kernel 'hqflashrd 0x90200000 0x600000 0x1406000'
setenv load_dtb 'hqflashrd 0x90100000 0x500000 0x3000'
setenv load_fs 'hqflashrd 0x93000000 0x1B00000 0x1600000'
setenv bootargs 'console=ttyAMA1,115200 earlycon=pl011,0x28001000  init=/linuxrc root=/dev/ram0 rw rootwait ramdisk_size=0x2000000 ksyslog'   
printenv
saveenv
```

```shell

//飞腾原生的
qspi0: spi@28008000 {

	compatible = "phytium,qspi-nor";

	reg = <0x0 0x28008000 0x0 0x1000>,<0x0 0x0 0x0 0x0fffffff>;

			reg-names = "qspi", "qspi_mm";

			clocks = <&sysclk_300mhz>;

			status = "disabled";

			#address-cells = <1>;

			#size-cells = <0>;

			flash@0 {

				compatible = "jedec,spi-nor";

				reg = <0x00>;

				spi-rx-bus-width = <1>;

				spi-max-frequency = <20000000>;

			};
};

//麒麟提供的

qspi@28014000 {
			compatible = "phytium,qspi";
			reg = <0x0 0x28014000 0x0 0x1000 0x0 0x0 0x0 0x20000000>;
			reg-names = "qspi", "qspi_mm";
			clocks = <0x4>;
			status = "ok";
			#address-cells = <0x1>;
			#size-cells = <0x0>;

			flash@0 {
				reg = <0x0>;
				spi-rx-bus-width = <0x4>;
				spi-max-frequency = <0x23c34600>;

				partitions {
					compatible = "fixed-partitions";

					#size-cells = <0x1>;

						label = "uboot";
```
