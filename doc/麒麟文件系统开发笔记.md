# 根文件系统打包

```shell

xz -C crc32 --lzma2=dict=8MiB rootf.cpio

xz -vv -l rootfs.cpio.xz

mkimage -A arm -O linux -T ramdisk -C lzma -n "ramdisk" -d rootfs.cpio.xz rootfs-current-20250530.xz.img

mkimage -A arm64 -O linux -C none -T kernel -a 0x80080000 -e 0x80080000 -d Image ./uImage

```
